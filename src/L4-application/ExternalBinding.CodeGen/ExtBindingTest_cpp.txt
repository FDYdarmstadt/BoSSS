int main(int argc, char** argv) {
    if (argc < 2 || argc > 3) { fprintf(stderr, "usage: ExtBindingTest <managed directory> [managed c degree]\n"); return 2; }
    try {
        // Match foam-extend's hardware trap settings, including before CLR startup.
        const int traps = FE_DIVBYZERO | FE_INVALID | FE_OVERFLOW;
        feenableexcept(traps);
        BoSSS::Globals::Init(argv[1]);
        assert((fegetexcept() & traps) == traps);
        Initializer init;
        init.BoSSSInitialize();
        InitMeshTest(argc == 3 ? std::stoi(argv[2]) : 2);
        assert((fegetexcept() & traps) == traps);
        init.BoSSSFinalize();
        assert((fegetexcept() & traps) == traps);
        return 0;
    } catch (const std::exception& error) {
        fprintf(stderr, "%s\n", error.what());
        return 1;
    }
}
